A `useConfirm()`{lang=ts} composable wraps [useModal()](/api/composables/use-modal), so `await confirm('Delete this item?')`{lang=ts} gives `true`{lang=ts} or `false`{lang=ts}. The answer comes once the modal has closed: Confirm, Cancel, `Esc` and a click outside all end in `closed`{lang=ts}. `open()`{lang=ts} always resolves, with `'opened'`{lang=ts} or [a message](/api/composables/use-modal#what-open-and-close-resolve-with), so a confirm that did not open answers `false`{lang=ts}.

## Preview

::code-preview
  :modal-await-confirm-preview

#code
  ```vue [Preview.vue]
    <script setup lang="ts">
    const confirm = useConfirm()
    const result = ref('')

    async function deleteItem() {
      result.value = await confirm('Delete this item?') ? 'Deleted' : 'Kept'
    }
    </script>

    <template>
      <VButton @click="deleteItem">
        Delete Item
      </VButton>
      <p v-if="result">
        {{ result }}
      </p>
    </template>
  ```

## `useConfirm()` composable

::code-group
  ```ts [composables/useConfirm.ts]
    import { useModal } from 'vue-final-modal'
    import ModalAwaitConfirm from '~/components/content/ModalAwaitConfirm.vue'

    export function useConfirm() {
      const title = ref('')
      let confirmed = false
      let answer: (confirmed: boolean) => void = () => {}

      const { open, close } = useModal({
        component: ModalAwaitConfirm,
        attrs: {
          title,
          onConfirm() {
            confirmed = true
            close()
          },
          onCancel() {
            close()
          },
          onClosed() {
            answer(confirmed)
          },
        },
      })

      return async function confirm(message: string): Promise<boolean> {
        title.value = message
        confirmed = false
        const answered = new Promise<boolean>(resolve => answer = resolve)
        if (await open() !== 'opened')
          return false
        return answered
      }
    }
  ```

  ```vue [ModalAwaitConfirm.vue]
    <script setup lang="ts">
    import { useId } from 'vue'
    import { VueFinalModal } from 'vue-final-modal'

    defineProps<{
      title: string
    }>()

    const emit = defineEmits<{
      (e: 'confirm'): void
      (e: 'cancel'): void
    }>()

    const titleId = useId()
    </script>

    <template>
      <VueFinalModal
        class="flex justify-center items-center"
        content-class="flex flex-col max-w-xl mx-4 p-4 bg-white dark:bg-gray-900 border dark:border-gray-700 rounded-lg space-y-2"
        content-transition="vfm-fade"
        overlay-transition="vfm-fade"
        :aria-labelledby="titleId"
      >
        <h1 :id="titleId" class="text-xl">
          {{ title }}
        </h1>
        <div class="flex justify-end gap-2">
          <button type="button" class="px-2 border rounded-lg" @click="emit('cancel')">
            Cancel
          </button>
          <button type="button" class="px-2 border rounded-lg" @click="emit('confirm')">
            Confirm
          </button>
        </div>
      </VueFinalModal>
    </template>
  ```
